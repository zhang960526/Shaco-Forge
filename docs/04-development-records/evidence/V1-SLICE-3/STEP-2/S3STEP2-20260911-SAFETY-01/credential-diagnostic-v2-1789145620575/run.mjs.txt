const alive = pid => { try { process.kill(pid, 0); return true } catch (error) { if (error.code === 'ESRCH') return false; throw error } }
async function until(predicate, label, timeout = 90_000) {
  const deadline = Date.now() + timeout
  while (Date.now() < deadline) { const value = await predicate(); if (value) return value; await delay(150) }
  throw new Error(`${label}: TIMEOUT`)
}
async function discovery() {
  return await until(async () => {
    try { return (await lifecycleRequest(paths.nativeHelper, 'discover')).status }
    catch (error) { if (error.message === 'BUSY') return undefined; throw error }
  }, 'discovery', 5_000)
}
async function desktop(index) {
  const control = join(root, `control-${index}.json`)
  const output = join(root, `snapshot-${index}.json`)
  const env = { ...process.env, SHACO_FORGE_WORKER_NODE: process.execPath, SHACO_FORGE_WORKER_ENTRY: paths.workerEntry,
    SHACO_FORGE_NATIVE_HELPER: paths.nativeHelper, SHACO_FORGE_DSH_HOME: dshHome,
    // This explicit evidence driver reads the full observer's bounded metadata.
    SHACO_FORGE_EVIDENCE_OBSERVER: '1',
    SHACO_FORGE_STEP2_CONTROL: control, SHACO_FORGE_STEP2_OUTPUT: output, SHACO_FORGE_STEP2_WORKSPACE: workspace }
  for (const key of ['SHACO_FORGE_USER_LOOP', 'SHACO_FORGE_EVIDENCE_PATH', 'SHACO_FORGE_EVIDENCE_INJECT_CARRIER_FAILURE', 'ELECTRON_RUN_AS_NODE']) delete env[key]
  const child = spawn(electron, [`--user-data-dir=${profile}`, "D:\\Project\\Shaco-Forge\\dist\\credential-diagnostic-v2\\main.mjs"], { env, windowsHide: true, stdio: ['ignore', 'pipe', 'pipe'] })
  children.push(child)
  let diagnostics = ''
  child.stdout.resume()
  child.stderr.on('data', chunk => { diagnostics = (diagnostics + chunk.toString()).slice(-8000) })
  let commandId = 0
  const read = async () => {
    if (child.exitCode !== null || child.signalCode !== null) throw new Error(`DESKTOP_EARLY_EXIT:${child.exitCode ?? child.signalCode}`)
    const value = await readFile(output, 'utf8').then(JSON.parse).catch(() => undefined)
    if (value?.failure) throw new Error(value.failure)
    if (value?.recovery?.connectionState === 'FAILED' || value?.recovery?.connectionState === 'INCOMPATIBLE') throw new Error(`PRODUCT_RECOVERY_FAILED:${value.recovery.failure}`)
    return value
  }
  const connected = async (generation, requireSession = false) => await until(async () => {
    const value = await read()
    return value?.recovery?.connectionState === 'CONNECTED' && value.recovery.generation >= generation
      && value.renderer?.projectionVisible && (!requireSession || value.readProof.sessionSnapshot) ? value : undefined
  }, 'Product CONNECTED')
  const command = async action => {
    const id = ++commandId
    await writeFile(control, JSON.stringify({ id, action }), 'utf8')
    if (action === 'close') return
    return await until(async () => { const value = await read(); return value?.commandId === id ? value.response : undefined }, action, 30_000)
  }
  return { child, connected, command, read, diagnostics: () => diagnostics }
}
function capture(label, value) {
  snapshots.push({ label, ...value })
  chain.push(label)
  if (!authorities.some(item => item.worker.pid === value.authority.worker.pid)) authorities.push(value.authority)
  assert.equal(value.renderer.requireAvailable, false)
  assert.equal(value.renderer.processAvailable, false)
  assert.equal(value.renderer.promptRuns, 0)
  assert.equal(value.recovery.authenticatedCarrier, true)
  assert.equal(value.recovery.clientGenerationReady, true)
  assert.equal(value.recovery.projectionRebuildComplete, true)
  for (const state of value.transitions) if (state.connectionState === 'CONNECTED') assert.ok(state.authenticatedCarrier && state.clientGenerationReady && state.projectionRebuildComplete)
}
