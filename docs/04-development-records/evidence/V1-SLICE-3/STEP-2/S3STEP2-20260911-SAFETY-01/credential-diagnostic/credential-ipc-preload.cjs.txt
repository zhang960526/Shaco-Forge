// Explicit non-shipped diagnostic only. Observe completed unset results; never
// read/record credential values, change arguments, or change the returned Promise.
const electron = require('electron')
if (electron.ipcMain) {
  const fs = require('node:fs')
  const path = require('node:path')
  const ipc = electron.ipcMain
  const original = ipc.handle.bind(ipc)
  ipc.handle = (channel, handler) => original(channel, (...args) => {
    const result = handler(...args)
    if (channel === 'transport:fetch' && args[1]?.url?.endsWith('/credentials/unset')) {
      result.then(response => {
        const destination = path.join(process.env.SHACO_FORGE_STEP2_EVIDENCE_ROOT, 'diagnostic-unset-ipc.json')
        const envelope = JSON.parse(response.body)
        fs.writeFileSync(destination, JSON.stringify({ result: envelope.result, explicitNonShippedDiagnostic: true, providerRuns: 0 }, null, 2) + '\n', 'utf8')
      }, error => {
        fs.writeFileSync(path.join(process.env.SHACO_FORGE_STEP2_EVIDENCE_ROOT, 'diagnostic-unset-ipc-error.json'), JSON.stringify({ error: error.message, explicitNonShippedDiagnostic: true }, null, 2) + '\n', 'utf8')
      })
    }
    return result
  })
}
